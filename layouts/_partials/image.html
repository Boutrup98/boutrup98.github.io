{{- /* Responsive, lazy-loaded WebP from an image resource. Call with: (dict "resource" $img "alt" "text" "class" "content-image") */ -}} {{- $img := .resource -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "content-image" -}} {{- $targets := slice 480 768 1200 -}} {{- $variants := slice -}} {{- range $w := $targets -}} {{- if le $w $img.Width -}} {{- $variants = $variants | append ($img.Resize (printf "%dx webp q80" $w)) -}} {{- end -}} {{- end -}} {{- /* Original narrower than the smallest target: emit a single WebP at native width. */ -}} {{- if eq (len $variants) 0 -}} {{- $variants = $variants | append ($img.Resize (printf "%dx webp q80" $img.Width)) -}} {{- end -}} {{- $largest := index $variants (sub (len $variants) 1) -}} {{- $srcset := slice -}} {{- range $v := $variants -}} {{- $srcset = $srcset | append (printf "%s %dw" $v.RelPermalink $v.Width) -}} {{- end -}}
